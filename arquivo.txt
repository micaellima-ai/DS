<?php
require_once __DIR__ . "/helpdesk-func.php";

$mensagem = "";

// IDENTIFICAR A OPERAÇÃO
if ($_SERVER["REQUEST_METHOD"] === "POST") {

    $acao = $_POST["acao"] ?? "";

    // CADASTRAR
    if ($acao === "cadastrar") {

        $nome = trim($_POST["nome"] ?? "");
        $setor = $_POST["setor"] ?? "";
        $equipamento = $_POST["equipamento"] ?? "";
        $descricao = trim($_POST["descricao"] ?? "");
        $prioridade = $_POST["prioridade"] ?? "";

        if ($nome === "" || $descricao === "") {
            $mensagem = "Preencha o nome e a descrição do problema.";
        } else {
            $resultado = cadastrarChamado(
                $nome,
                $setor,
                $equipamento,
                $descricao,
                $prioridade
            );

            $mensagem = $resultado
                ? "Chamado cadastrado com sucesso!"
                : "Erro ao salvar o chamado.";
        }
    }

    // ATUALIZAR STATUS
    elseif ($acao === "atualizar") {

        $posicao = filter_var(
            $_POST["posicao"] ?? "",
            FILTER_VALIDATE_INT
        );

        $novoStatus = $_POST["status"] ?? "";

        if ($posicao === false || $posicao === null || $posicao < 0) {
            $mensagem = "Número do chamado inválido.";
        } elseif (atualizarChamado($posicao, $novoStatus)) {
            $mensagem = "Status atualizado com sucesso!";
        } else {
            $mensagem = "Não foi possível atualizar o chamado.";
        }
    }

    // EXCLUIR
    elseif ($acao === "deletar") {

        $posicao = filter_var(
            $_POST["posicao"] ?? "",
            FILTER_VALIDATE_INT
        );

        if ($posicao === false || $posicao === null || $posicao < 0) {
            $mensagem = "Número do chamado inválido.";
        } elseif (excluirChamado($posicao)) {
            $mensagem = "Chamado excluído com sucesso!";
        } else {
            $mensagem = "Chamado não encontrado.";
        }
    }
}

// CONSULTAR CHAMADOS E RELATÓRIO
$chamados = listarChamados();
$relatorio = contarChamados();

        <textarea name="descricao" required></textarea>
        <br><br>

        <label>Prioridade:</label>
        <select name="prioridade" required>
            <option value="Baixa">Baixa</option>
            <option value="Média">Média</option>
            <option value="Alta">Alta</option>
        </select>
        <br><br>

        <button type="submit" name="acao" value="cadastrar">
            Cadastrar chamado
        </button>

    </form>
